import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        System.out.println("Enter the array value : ");
        int n = s.nextInt();
        System.out.println("Original array : ");
        int a[] = new int[n];
        for(int i=0;i<n;i++)
        {
            a[i] = s.nextInt();
        }
        
        for(int i=0;i<n;i++)
        {
            int num = a[i];
            int div = num/2;
            System.out.println("The div value : "+ div);
            if(div%2==0)
            {
                int even = (div+10)*2;
                System.out.println("The added even no : "+even);
            if(even>20)
            {
                System.out.println("High");
            }
            else
            {
                System.out.println("Invalid");
            }
            }
            else
            {
                int odd = (div-5)*3;
                System.out.println("The added odd no :"+ odd);
                if(odd>20)
                {
                System.out.println("low");
                
                }
                else
                {
                    System.out.println("hugee");
                }
            }
            
            
        }
        
        
    }
}
