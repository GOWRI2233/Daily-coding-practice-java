class Main {
    public static void main(String[] args) {
        int a[] = {5,12,8,15,20,3,11};
        int count = 0;
        for(int i=0;i<a.length;i++)
        {
            if(a[i]>=10)
            {
                count = count + 1;   
            }
        }
        System.out.println(count);
    }
}
