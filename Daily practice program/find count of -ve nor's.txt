class Main {
    public static void main(String[] args) {
        int a[] = {1,-1,3,-10,5,-4,9,-7,0,-3};
        int count = 0;
        for(int i=0;i<a.length;i++)
        {
            if(a[i]<0)
            {
                count = count + 1;
            }
        }
        System.out.println(count);
    }
}
