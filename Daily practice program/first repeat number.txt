import java.util.*;
public class Main
{
	public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        System.out.println("Enter the size : ");
        int n = s.nextInt();
        System.out.println("Enter the array value : ");
        int a[] = new int[n];
        for(int i=0;i<n;i++)
        {
            a[i] = s.nextInt();
        }
        boolean found = false;
        for(int i=0;i<n;i++)
        {
            for(int j=i+1;j<n;j++)  
            {
                if(a[i]==a[j])
                {
                    System.out.println(a[i]);
                    found=true;
                    break;
                }
            }
            if(found)
            {
                break;
            }
        }
        if(!found)
        {
            System.out.println("Not repeat");
        }
	}
}
