class Main
{
    public static void main(String args[])
    {
        int a[] = {1,2,3,4};
        int a1[] = new int[a.length];
        int j=0;
        for(int i=0;i<a.length;i++)
        {
            if(a[i]%2==0)
            {
                a1[j] = a[i];
                j++;
            }
        }
        
        for(int i=0;i<j;i++)
        {
            System.out.print(a1[i]+ " ");
        }
        int sum = 0;
        for(int i=0;i<j;i++)
        {
            sum = sum+a1[i];
        }
        System.out.print("The sum is : "+sum);
    }
}
