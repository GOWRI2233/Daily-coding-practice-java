public class Main
{
	public static void main(String[] args) {
		String s = "1234";
	    int sum = 0;
		for(int i=0;i<s.length();i++)
		{
		    int digit = s.charAt(i) - '0';
		    int position = i + 1;
		    int value = position * digit;
		    sum = sum + value;
		}
		System.out.println(sum);
	}
}



public class Main
{
    public static void main(String args[])
    {
        String s = "2465";
        int score = 0;
        for(int i=0;i<s.length();i++)
        {
            int digit = s.charAt(i) - '0';
            int position = i+1;
            int value = position * digit;
            if(value%2==0)
            {
                score = score + digit;
            }
            else
            {
                score = score - digit;
            }
        }
        if(s.contains("5"))
        {
            score = score * 3;
        }
        System.out.println(score);
    }
}
