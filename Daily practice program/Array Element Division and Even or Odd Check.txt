import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        System.out.println("Enter the array value : ");
        int n = s.nextInt();
        System.out.println("Original array : ");
        int a[] = new int[n];
        for(int i=0;i<n;i++)
        {
            a[i] = s.nextInt();
        }
        
        for(int i=0;i<n;i++)
        {
            int num = a[i];
            int div = num/2;
            if(div%2==0)
            {
                System.out.println(div + " is even number");
            }
            else
            {
                System.out.println(div + " is odd number");
            }
            
        }
        
        
    }
}
